% Секции без номеров (введение, заключение...), вместо section*{}
\newcommand{\anonsection}[1]{
    \newpage
    \phantomsection % Корректный переход по ссылкам в содержании
    \paragraph{\textbf{\centering\MakeTextUppercase{{#1}}}}
    \addcontentsline{toc}{section}{#1}
}

% Секции для приложений
% Перед приложениями задаем счетчик section \setcounter{section}{1}
\newcommand{\appsection}[1]{
    \renewcommand{\thesection}{\Asbuk{section}} % Перевод нумерации заголовков в вид букв А, Б, В...
    \setcounter{figure}{0}
    \setcounter{lstlisting}{0}
    \setcounter{table}{0}
    \newpage
    \phantomsection
    \refstepcounter{section}
    \paragraph{\centerline{\textbf{Приложение \thesection}}}
    \centerline{#1} \hfill % Название приложения
    \addcontentsline{toc}{section}{{Приложение \thesection. #1}} % Прописывается в содержание
}
