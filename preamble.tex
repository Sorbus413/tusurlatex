}

% приложения
\newcommand{\appsection}[1]{%
    \newpage
    \phantomsection   
    \stepcounter{AppCount}
    \setcounter{figure}{0}
    \setcounter{table}{0}
    \setcounter{lstlisting}{0}
    \renewcommand{\thefigure}{\theAppCount.\arabic{figure}}
    \renewcommand{\thetable}{\theAppCount.\arabic{table}}
    \renewcommand{\thelstlisting}{\theAppCount.\arabic{lstlisting}}
    \begin{center}
        \textbf{ПРИЛОЖЕНИЕ \theAppCount} \\
        \textbf{#1}
    \end{center}
    \addcontentsline{toc}{section}{Приложение \theAppCount. #1} 
}
